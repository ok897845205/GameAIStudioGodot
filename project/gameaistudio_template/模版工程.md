### scripts/player.gd
**严格使用 Godot 4 API，禁止 Godot 3 写法：**

```gdscript
extends CharacterBody2D

const SPEED: float = 200.0
const JUMP_VELOCITY: float = -400.0
const GRAVITY: float = 980.0

@export var max_jump_count: int = 2

var jump_count: int = 0

func _physics_process(delta: float) -> void:
    # 重力
    if not is_on_floor():
        velocity.y += GRAVITY * delta
    else:
        jump_count = 0

    # 跳跃
    if Input.is_action_just_pressed("ui_accept"):
        jump()

    # 水平移动
    var direction: float = Input.get_axis("ui_left", "ui_right")
    velocity.x = direction * SPEED

    move_and_slide()  # Godot 4：无参数，返回 bool，不赋回 velocity

func jump() -> void:
    if jump_count < max_jump_count:
        velocity.y = JUMP_VELOCITY
        jump_count += 1
```

**禁止写法（Godot 3 残留）：**
- `velocity = move_and_slide(velocity, Vector2.UP)` ❌
- `move_and_slide(velocity, Vector2.UP)` ❌
- `yield(...)` ❌
- `export var` ❌

### scripts/game_manager.gd
```gdscript
extends Node

signal game_started
signal game_over(score: int)

var score: int = 0
var is_game_running: bool = false

func start_game() -> void:
    score = 0
    is_game_running = true
    game_started.emit()

func add_score(points: int) -> void:
    score += points

func end_game() -> void:
    is_game_running = false
    game_over.emit(score)
```

### scripts/enemy.gd
```gdscript
extends CharacterBody2D

const SPEED: float = 80.0
const GRAVITY: float = 980.0

@export var patrol_distance: float = 100.0

var direction: float = 1.0
var start_position: Vector2

func _ready() -> void:
    start_position = global_position

func _physics_process(delta: float) -> void:
    if not is_on_floor():
        velocity.y += GRAVITY * delta

    velocity.x = direction * SPEED

    if abs(global_position.x - start_position.x) >= patrol_distance:
        direction *= -1.0

    move_and_slide()
```

### tests/test_player.gd
```gdscript
extends GutTest

var player_scene: PackedScene = preload("res://scenes/main.tscn")
var player: CharacterBody2D

func before_each() -> void:
    # 直接实例化脚本进行单元测试
    player = CharacterBody2D.new()
    player.set_script(load("res://scripts/player.gd"))
    add_child_autofree(player)

func test_initial_jump_count_is_zero() -> void:
    assert_eq(player.jump_count, 0, "初始跳跃次数应为 0")

func test_max_jump_count_default() -> void:
    assert_eq(player.max_jump_count, 2, "默认最大跳跃次数应为 2")

func test_jump_increments_count() -> void:
    player.jump_count = 0
    player.jump()
    assert_eq(player.jump_count, 1, "跳跃后计数应增加")

func test_jump_respects_max_count() -> void:
    player.jump_count = 2
    var velocity_before: float = player.velocity.y
    player.jump()
    assert_eq(player.velocity.y, velocity_before, "超出最大跳跃次数后速度不应改变")
```

### tools/ci/validate_project.gd
```gdscript
## GameAIStudio headless 验证脚本
## 调用方式：
##   Godot_console.exe --headless --path . -s tools/ci/validate_project.gd
extends SceneTree

func _init() -> void:
    print("[GameAIStudio] 开始项目验证...")

    var errors: Array[String] = []

    # 检查必要文件
    var required_files: Array[String] = [
        "res://scenes/main.tscn",
        "res://scripts/player.gd",
        "res://scripts/game_manager.gd",
        "res://addons/gut/plugin.cfg",
    ]
    for path in required_files:
        if not FileAccess.file_exists(path):
            errors.append("缺少必要文件：" + path)

    if errors.is_empty():
        print("[GameAIStudio] 验证通过 ✅")
        quit(0)
    else:
        for err in errors:
            printerr("[GameAIStudio] 错误：" + err)
        quit(1)
```
